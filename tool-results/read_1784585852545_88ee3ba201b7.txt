   114→// ---------- Helpers ----------

   116→function formatDate(dateStr: string) {
   117→  const d = new Date(dateStr);
   118→  return d.toLocaleDateString('en-IN', {
   119→    day: '2-digit',
   120→    month: 'short',
   121→    year: 'numeric',
   122→  });
   123→}
   124→
   125→function calcBusinessDays(start: string, end: string): number {
   126→  const s = new Date(start);
   127→  const e = new Date(end);
   128→  if (e < s) return 0;
   129→  const diffMs = Math.abs(e.getTime() - s.getTime());
   130→  return Math.ceil(diffMs / (1000 * 60 * 60 * 24)) + 1;
   131→}
   132→
   133→function getBalanceColor(available: number, allocated: number): string {
   134→  if (allocated <= 0) return 'text-muted-foreground';
   135→  const pct = (available / allocated) * 100;
   136→  if (pct > 50) return 'text-emerald-600 dark:text-emerald-400';
   137→  if (pct >= 25) return 'text-amber-600 dark:text-amber-400';
   138→  return 'text-red-600 dark:text-red-400';
   139→}
   140→
   141→function getProgressColor(available: number, allocated: number): string {
   142→  if (allocated <= 0) return '[&>[data-slot=progress-indicator]]:bg-muted-foreground';
   143→  const pct = (available / allocated) * 100;
   144→  if (pct > 50) return '[&>[data-slot=progress-indicator]]:bg-emerald-500';
   145→  if (pct >= 25) return '[&>[data-slot=progress-indicator]]:bg-amber-500';
   146→  return '[&>[data-slot=progress-indicator]]:bg-red-500';
   147→}
   148→
   149→function getBalanceBorderColor(available: number, allocated: number): string {
   150→  if (allocated <= 0) return 'border-border';
   151→  const pct = (available / allocated) * 100;
   152→  if (pct > 50) return 'border-emerald-200 dark:border-emerald-800';
   153→  if (pct >= 25) return 'border-amber-200 dark:border-amber-800';
   154→  return 'border-red-200 dark:border-red-800';
   155→}
