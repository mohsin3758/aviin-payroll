   160→   160→/* ------------------------------------------------------------------ */
   161→   161→/*  Component                                                          */
   162→   162→/* ------------------------------------------------------------------ */
   163→   163→
   164→   164→export default function PayrollView() {
   165→   165→  const { refreshKey } = usePayrollStore();
   166→   166→
   167→   167→  // Header state
   168→   168→  const now = new Date();
   169→   169→  const [month, setMonth] = useState(String(now.getMonth() + 1));
   170→   170→  const [year, setYear] = useState(String(now.getFullYear()));
   171→   171→
   172→   172→  // Processing dialog state
   173→   173→  const [confirmOpen, setConfirmOpen] = useState(false);
   174→   174→  const [processing, setProcessing] = useState(false);
   175→   175→
   176→   176→  // Table state
   177→   177→  const [runs, setRuns] = useState<PayrollRunSummary[]>([]);
   178→   178→  const [loadingRuns, setLoadingRuns] = useState(true);
   179→   179→
   180→   180→  // Detail dialog state
   181→   181→  const [detailOpen, setDetailOpen] = useState(false);
   182→   182→  const [selectedRun, setSelectedRun] = useState<PayrollRunDetail | null>(null);
   183→   183→  const [loadingDetail, setLoadingDetail] = useState(false);
   184→   184→  const [statusUpdating, setStatusUpdating] = useState(false);
   185→   185→
   186→   186→  /* --- Fetch payroll runs --- */
   187→   187→  const fetchRuns = useCallback(async () => {
   188→   188→    setLoadingRuns(true);
   189→   189→    try {
   190→   190→      const res = await fetch('/api/payroll');
   191→   191→      if (!res.ok) throw new Error('Failed to fetch payroll runs');
   192→   192→      const data: PayrollRunSummary[] = await res.json();
   193→   193→      setRuns(data);
   194→   194→    } catch {
   195→   195→      toast.error('Failed to load payroll runs');
   196→   196→    } finally {
   197→   197→      setLoadingRuns(false);
   198→   198→    }
   199→   199→  }, []);
   200→   200→
   201→   201→  useEffect(() => {
   202→   202→    fetchRuns();
   203→   203→  }, [fetchRuns, refreshKey]);
   204→   204→
   205→   205→  /* --- Process payroll --- */
   206→   206→  const handleProcess = async () => {
   207→   207→    setProcessing(true);
   208→   208→    try {
   209→   209→      const res = await fetch('/api/payroll', {
   210→   210→        method: 'POST',
   211→   211→        headers: { 'Content-Type': 'application/json' },
   212→   212→        body: JSON.stringify({ month: Number(month), year: Number(year) }),
   213→   213→      });
   214→   214→      const data = await res.json();
   215→   215→      if (!res.ok) throw new Error(data.error || 'Processing failed');
   216→   216→
   217→   217→      toast.success(
   218→   218→        `Payroll processed for ${MONTHS[Number(month) - 1]} ${year} — ${data.totalEmployees} employees`
   219→   219→      );
   220→   220→      setConfirmOpen(false);
   221→   221→      fetchRuns();
   222→   222→    } catch (err) {
   223→   223→      toast.error(err instanceof Error ? err.message : 'Processing failed');
   224→   224→    } finally {
   225→   225→      setProcessing(false);
   226→   226→    }
   227→   227→  };
   228→   228→
   229→   229→  /* --- View detail --- */
   230→   230→  const openDetail = async (runId: string) => {
   231→   231→    setDetailOpen(true);
   232→   232→    setLoadingDetail(true);
   233→   233→    setSelectedRun(null);
   234→   234→    try {
   235→   235→      const res = await fetch(`/api/payroll/${runId}`);
   236→   236→      if (!res.ok) throw new Error('Failed to fetch details');
   237→   237→      const data: PayrollRunDetail = await res.json();
   238→   238→      setSelectedRun(data);
   239→   239→    } catch {
   240→   240→      toast.error('Failed to load payroll details');
   241→   241→      setDetailOpen(false);
   242→   242→    } finally {
   243→   243→      setLoadingDetail(false);
   244→   244→    }
   245→   245→  };
   246→   246→
   247→   247→  /* --- Status transition --- */
   248→   248→  const handleStatusChange = async (runId: string, newStatus: string) => {
   249→   249→    setStatusUpdating(true);
   250→   250→    try {
   251→   251→      const res = await fetch(`/api/payroll/${runId}`, {
   252→   252→        method: 'PUT',
   253→   253→        headers: { 'Content-Type': 'application/json' },
   254→   254→        body: JSON.stringify({ status: newStatus }),
   255→   255→      });
   256→   256→      const data = await res.json();
   257→   257→      if (!res.ok) throw new Error(data.error || 'Status update failed');
   258→   258→
   259→   259→      toast.success(`Payroll marked as "${newStatus}"`);
   260→   260→      // Refresh detail view
   261→   261→      setSelectedRun(data);
   262→   262→      fetchRuns();
   263→   263→    } catch (err) {
   264→   264→      toast.error(err instanceof Error ? err.message : 'Status update failed');
   265→   265→    } finally {
   266→   266→      setStatusUpdating(false);
   267→   267→    }
   268→   268→  };
   269→   269→
   270→   270→  /* --- Detail totals --- */
   271→   271→  const detailTotals = (detail: PayrollRunDetail) => {
   272→   272→    const t = {
   273→   273→      paidDays: 0, basic: 0, dearnessAllowance: 0, houseRentAllowance: 0,
   274→   274→      specialAllowance: 0, totalEarnings: 0,
   275→   275→      employeePF: 0, employeeESI: 0, tds: 0, professionalTax: 0, lwf: 0,
   276→   276→      totalDeductions: 0, netSalary: 0, employerPF: 0, employerESI: 0,
   277→   277→    };
   278→   278→    for (const d of detail.details) {
   279→   279→      t.paidDays += d.paidDays;
   280→   280→      t.basic += d.basic;
   281→   281→      t.dearnessAllowance += d.dearnessAllowance;
   282→   282→      t.houseRentAllowance += d.houseRentAllowance;
   283→   283→      t.specialAllowance += d.specialAllowance;
   284→   284→      t.totalEarnings += d.totalEarnings;
   285→   285→      t.employeePF += d.employeePF;
   286→   286→      t.employeeESI += d.employeeESI;
   287→   287→      t.tds += d.tds;
   288→   288→      t.professionalTax += d.professionalTax;
   289→   289→      t.lwf += d.lwf;
   290→   290→      t.totalDeductions += d.totalDeductions;
   291→   291→      t.netSalary += d.netSalary;
   292→   292→      t.employerPF += d.employerPF;
   293→   293→      t.employerESI += d.employerESI;
   294→   294→    }
   295→   295→    return t;
   296→   296→  };
   297→   297→
   298→   298→  return (
   299→   299→    <div className="space-y-6">
   300→   300→      {/* ── Header ─────────────────────────────────────────────────── */}
   301→   301→      <div className="flex flex-col gap-4 sm:flex-row sm:items-end sm:justify-between">
   302→   302→        <div>
   303→   303→          <h1 className="text-2xl font-bold tracking-tight">Payroll Processing</h1>
   304→   304→          <p className="text-muted-foreground text-sm mt-1">
   305→   305→            Process monthly payroll and manage run statuses
   306→   306→          </p>
   307→   307→        </div>
   308→   308→
   309→   309→        <div className="flex flex-wrap items-end gap-3">
   310→   310→          <div className="space-y-1.5">
   311→   311→            <Label htmlFor="payroll-month">Month</Label>
   312→   312→            <Select value={month} onValueChange={setMonth}>
   313→   313→              <SelectTrigger id="payroll-month" className="w-[160px]">
   314→   314→                <SelectValue />
   315→   315→              </SelectTrigger>
   316→   316→              <SelectContent>
   317→   317→                {MONTHS.map((m, i) => (
   318→   318→                  <SelectItem key={i} value={String(i + 1)}>
   319→   319→                    {m}
   320→   320→                  </SelectItem>
   321→   321→                ))}
   322→   322→              </SelectContent>
   323→   323→            </Select>
   324→   324→          </div>
   325→   325→
   326→   326→          <div className="space-y-1.5">
   327→   327→            <Label htmlFor="payroll-year">Year</Label>
   328→   328→            <Input
   329→   329→              id="payroll-year"
   330→   330→              type="number"
   331→   331→              min={2000}
   332→   332→              max={2100}
   333→   333→              value={year}
   334→   334→              onChange={(e) => setYear(e.target.value)}
   335→   335→              className="w-[100px]"
   336→   336→            />
   337→   337→          </div>
   338→   338→
   339→   339→          <AlertDialog open={confirmOpen} onOpenChange={setConfirmOpen}>
   340→   340→            <AlertDialogTrigger asChild>
   341→   341→              <Button className="bg-emerald-600 hover:bg-emerald-700 text-white">
   342→   342→                <Calculator className="size-4" />
   343→   343→                Process Payroll
   344→   344→              </Button>
   345→   345→            </AlertDialogTrigger>
   346→   346→            <AlertDialogContent>
   347→   347→              <AlertDialogHeader>
   348→   348→                <AlertDialogTitle>Process Payroll</AlertDialogTitle>
   349→   349→                <AlertDialogDescription>
   350→   350→                  Process payroll for <span className="font-semibold">{MONTHS[Number(month) - 1]} {year}</span>?
   351→   351→                  This will calculate salary for all active employees based on their attendance.
   352→   352→                </AlertDialogDescription>
   353→   353→              </AlertDialogHeader>
   354→   354→              <AlertDialogFooter>
   355→   355→                <AlertDialogCancel disabled={processing}>Cancel</AlertDialogCancel>
   356→   356→                <AlertDialogAction
   357→   357→                  onClick={handleProcess}
   358→   358→                  disabled={processing}
   359→   359→                  className="bg-emerald-600 hover:bg-emerald-700 text-white"
   360→   360→                >
   361→   361→                  {processing ? (
   362→   362→                    <>
   363→   363→                      <Loader2 className="size-4 animate-spin" />
   364→   364→                      Processing…
   365→   365→                    </>
   366→   366→                  ) : (
   367→   367→                    'Confirm'
   368→   368→                  )}
   369→   369→                </AlertDialogAction>
   370→   370→              </AlertDialogFooter>
   371→   371→            </AlertDialogContent>
   372→   372→          </AlertDialog>
   373→   373→        </div>
   374→   374→      </div>
   375→   375→
   376→   376→      {/* ── Runs Table ─────────────────────────────────────────────── */}
   377→   377→      <Card>
   378→   378→        <CardHeader>
   379→   379→          <div className="flex items-center justify-between">
   380→   380→            <div>
   381→   381→              <CardTitle>Payroll Runs</CardTitle>
   382→   382→              <CardDescription>All processed payroll periods</CardDescription>
   383→   383→            </div>
   384→   384→            <Button variant="outline" size="sm" onClick={fetchRuns} disabled={loadingRuns}>
   385→   385→              <RefreshCw className={`size-4 ${loadingRuns ? 'animate-spin' : ''}`} />
   386→   386→              Refresh
   387→   387→            </Button>
   388→   388→          </div>
   389→   389→        </CardHeader>
   390→   390→        <CardContent>
   391→   391→          {loadingRuns ? (
   392→   392→            <div className="space-y-3">
   393→   393→              {Array.from({ length: 3 }).map((_, i) => (
   394→   394→                <Skeleton key={i} className="h-12 w-full" />
   395→   395→              ))}
   396→   396→            </div>
   397→   397→          ) : runs.length === 0 ? (
   398→   398→            <div className="flex flex-col items-center justify-center py-12 text-muted-foreground">
   399→   399→              <AlertCircle className="size-10 mb-3 opacity-40" />
   400→   400→              <p className="text-sm">No payroll runs found. Process your first payroll above.</p>
   401→   401→            </div>
   402→   402→          ) : (
   403→   403→            <div className="overflow-x-auto">
   404→   404→              <Table>
   405→   405→                <TableHeader>
   406→   406→                  <TableRow>
   407→   407→                    <TableHead>Month</TableHead>
   408→   408→                    <TableHead>Year</TableHead>
   409→   409→                    <TableHead className="text-right">Employees</TableHead>
   410→   410→                    <TableHead className="text-right">Total Gross</TableHead>
   411→   411→                    <TableHead className="text-right">Total Deductions</TableHead>
   412→   412→                    <TableHead className="text-right">Total Net</TableHead>
   413→   413→                    <TableHead className="text-right">Total PF</TableHead>
   414→   414→                    <TableHead className="text-right">Total ESI</TableHead>
   415→   415→                    <TableHead className="text-right">Total TDS</TableHead>
   416→   416→                    <TableHead>Status</TableHead>
   417→   417→                    <TableHead className="text-right">Actions</TableHead>
   418→   418→                  </TableRow>
   419→   419→                </TableHeader>
   420→   420→                <TableBody>
   421→   421→                  {runs.map((run) => (
   422→   422→                    <TableRow key={run.id}>
   423→   423→                      <TableCell className="font-medium">{MONTHS[run.month - 1]}</TableCell>
   424→   424→                      <TableCell>{run.year}</TableCell>
   425→   425→                      <TableCell className="text-right">{run.totalEmployees}</TableCell>
   426→   426→                      <TableCell className="text-right">{fmt(run.totalGrossSalary)}</TableCell>
   427→   427→                      <TableCell className="text-right">{fmt(run.totalDeductions)}</TableCell>
   428→   428→                      <TableCell className="text-right font-semibold">{fmt(run.totalNetSalary)}</TableCell>
   429→   429→                      <TableCell className="text-right">{fmt(run.totalEmployeePF)}</TableCell>
   430→   430→                      <TableCell className="text-right">{fmt(run.totalEmployeeESI)}</TableCell>
   431→   431→                      <TableCell className="text-right">{fmt(run.totalTDS)}</TableCell>
   432→   432→                      <TableCell>
   433→   433→                        <Badge
   434→   434→                          variant="outline"
   435→   435→                          className={statusColor[run.status] || ''}
   436→   436→                        >
   437→   437→                          {run.status.charAt(0).toUpperCase() + run.status.slice(1)}
   438→   438→                        </Badge>
   439→   439→                      </TableCell>
   440→   440→                      <TableCell className="text-right">
   441→   441→                        <Button
   442→   442→                          variant="ghost"
   443→   443→                          size="sm"
   444→   444→                          onClick={() => openDetail(run.id)}
   445→   445→                        >
   446→   446→                          <Eye className="size-4 mr-1" />
   447→   447→                          View
   448→   448→                        </Button>
   449→   449→                      </TableCell>
   450→   450→                    </TableRow>
   451→   451→                  ))}
   452→   452→                </TableBody>
   453→   453→              </Table>
   454→   454→            </div>
   455→   455→          )}
   456→   456→        </CardContent>
   457→   457→      </Card>
   458→   458→
   459→   459→      {/* ── Detail Dialog ──────────────────────────────────────────── */}
   460→   460→      <Dialog open={detailOpen} onOpenChange={setDetailOpen}>
   461→   461→        <DialogContent className="max-w-[95vw] lg:max-w-[90vw] xl:max-w-[1200px] max-h-[90vh] overflow-y-auto">
   462→   462→          <DialogHeader>
   463→   463→            <DialogTitle className="flex items-center gap-2">
   464→   464→              {selectedRun && (
   465→   465→                <>
   466→   466→                  Payroll Detail — {MONTHS[selectedRun.month - 1]} {selectedRun.year}
   467→   467→                  <Badge
   468→   468→                    variant="outline"
   469→   469→                    className={statusColor[selectedRun.status] || ''}
   470→   470→                  >
   471→   471→                    {selectedRun.status.charAt(0).toUpperCase() + selectedRun.status.slice(1)}
   472→   472→                  </Badge>
   473→   473→                </>
   474→   474→              )}
   475→   475→            </DialogTitle>
   476→   476→            <DialogDescription>
   477→   477→              Employee-wise salary breakdown
   478→   478→            </DialogDescription>
   479→   479→          </DialogHeader>
   480→   480→
   481→   481→          {loadingDetail ? (
   482→   482→            <div className="space-y-3 py-4">
   483→   483→              {Array.from({ length: 5 }).map((_, i) => (
   484→   484→                <Skeleton key={i} className="h-10 w-full" />
   485→   485→              ))}
   486→   486→            </div>
   487→   487→          ) : selectedRun ? (
   488→   488→            <div className="space-y-4">
   489→   489→              {/* Status action bar */}
   490→   490→              {nextStatus[selectedRun.status] && (
   491→   491→                <div className="flex items-center gap-3 p-3 rounded-lg bg-amber-50 border border-amber-200">
   492→   492→                  <ChevronLeft className="size-4 text-amber-600 shrink-0" />
