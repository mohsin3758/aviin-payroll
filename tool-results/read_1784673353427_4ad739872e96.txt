   120→   120→function formatTime(dateStr: string | null): string {
   121→   121→  if (!dateStr) return '--';
   122→   122→  const d = new Date(dateStr);
   123→   123→  return d.toLocaleTimeString('en-IN', { hour: '2-digit', minute: '2-digit', hour12: true });
   124→   124→}
   125→   125→
   126→   126→function formatDate(dateStr: string): string {
   127→   127→  const d = new Date(dateStr);
   128→   128→  return d.toLocaleDateString('en-IN', { day: '2-digit', month: 'short', year: 'numeric' });
   129→   129→}
   130→   130→
   131→   131→function nowInIST(): Date {
   132→   132→  const now = new Date();
   133→   133→  const istOffset = 5.5 * 60 * 60 * 1000;
   134→   134→  const utcOffset = now.getTimezoneOffset() * 60 * 1000;
   135→   135→  return new Date(now.getTime() + istOffset + utcOffset);
   136→   136→}
