   237→   237→function getInitials(firstName: string, lastName: string | null) {
   238→   238→  const f = firstName?.charAt(0)?.toUpperCase() || ''
   239→   239→  const l = lastName?.charAt(0)?.toUpperCase() || ''
   240→   240→  return (f + l) || '?'
   241→   241→}

   243→   243→function getFullName(emp: Employee) {
   244→   244→  return `${emp.firstName}${emp.lastName ? ' ' + emp.lastName : ''}`
   245→   245→}

   247→   247→function computeSalary(form: FormData) {
   248→   248→  const basic = parseFloat(form.basic) || 0
   249→   249→  const da = parseFloat(form.dearnessAllowance) || 0
   250→   250→  const hra = parseFloat(form.houseRentAllowance) || 0
   251→   251→  const conv = parseFloat(form.conveyanceAllowance) || 0
   252→   252→  const med = parseFloat(form.medicalAllowance) || 0
   253→   253→  const special = parseFloat(form.specialAllowance) || 0
   254→   254→  const gross = basic + da + hra + conv + med + special
   255→   255→  const employerPF = form.pfApplicable ? Math.round(basic * 0.12) : 0
   256→   256→  const employerESI = form.esiApplicable && gross <= 21000 ? Math.round(gross * 0.0325) : 0
   257→   257→  const monthlyCTC = gross + employerPF + employerESI
   258→   258→  return { gross, employerPF, employerESI, monthlyCTC }
   259→   259→}
