   116→   116→function formatDate(dateStr: string) {
   117→   117→  const d = new Date(dateStr);
   118→   118→  return d.toLocaleDateString('en-IN', {
   119→   119→    day: '2-digit',
   120→   120→    month: 'short',
   121→   121→    year: 'numeric',
   122→   122→  });
   123→   123→}
   124→   124→
   125→   125→function calcBusinessDays(start: string, end: string): number {
   126→   126→  const s = new Date(start);
   127→   127→  const e = new Date(end);
   128→   128→  if (e < s) return 0;
   129→   129→  const diffMs = Math.abs(e.getTime() - s.getTime());
   130→   130→  return Math.ceil(diffMs / (1000 * 60 * 60 * 24)) + 1;
   131→   131→}
   132→   132→
   133→   133→function getBalanceColor(available: number, allocated: number): string {
   134→   134→  if (allocated <= 0) return 'text-muted-foreground';
   135→   135→  const pct = (available / allocated) * 100;
   136→   136→  if (pct > 50) return 'text-emerald-600 dark:text-emerald-400';
   137→   137→  if (pct >= 25) return 'text-amber-600 dark:text-amber-400';
   138→   138→  return 'text-red-600 dark:text-red-400';
   139→   139→}
   140→   140→
   141→   141→function getProgressColor(available: number, allocated: number): string {
   142→   142→  if (allocated <= 0) return '[&>[data-slot=progress-indicator]]:bg-muted-foreground';
   143→   143→  const pct = (available / allocated) * 100;
   144→   144→  if (pct > 50) return '[&>[data-slot=progress-indicator]]:bg-emerald-500';
   145→   145→  if (pct >= 25) return '[&>[data-slot=progress-indicator]]:bg-amber-500';
   146→   146→  return '[&>[data-slot=progress-indicator]]:bg-red-500';
   147→   147→}
   148→   148→
   149→   149→function getBalanceBorderColor(available: number, allocated: number): string {

   155→   155→}
